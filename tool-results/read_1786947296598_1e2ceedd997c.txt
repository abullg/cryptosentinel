    66→/**
    67→ * Format an OpenRouter HTTP error into a clear, actionable message.
    68→ * Returns a string suitable for surfacing in the UI Activity feed.
    69→ *
    70→ * This is the single source of truth for "what does this status code mean
    71→ * for the user". Without it, callers just dumped the raw JSON body which
    72→ * was useless for end users trying to debug their config.
    73→ */
    74→export function formatOpenRouterError(status: number, body: string, model: string): string {
    75→  // Try to extract the OpenRouter error message
    76→  let upstreamMessage = '';
    77→  try {
    78→    const parsed = JSON.parse(body);
    79→    upstreamMessage = parsed?.error?.message || parsed?.message || '';
    80→  } catch {
    81→    upstreamMessage = body.slice(0, 200);
    82→  }
    83→
    84→  switch (status) {
    85→    case 400:
    86→      return `OpenRouter rejected the request (400 Bad Request${model ? ` for ${model}` : ''}). ${upstreamMessage}`.trim();
    87→    case 401:
    88→      return 'OpenRouter API key is invalid or missing (401). Get a valid key at https://openrouter.ai/keys — it must start with "sk-or-v1-".';
    89→    case 402:
    90→      return 'OpenRouter credits exhausted (402 Payment Required). Add credits at https://openrouter.ai/credits.';
    91→    case 403:
    92→      return `OpenRouter denied access (403 Forbidden). The key may not have permission to use ${model}. ${upstreamMessage}`.trim();
    93→    case 408:
    94→      return 'OpenRouter request timed out (408). The model may be overloaded — retry, or switch to a faster model.';
    95→    case 429:
    96→      return 'OpenRouter rate limit hit (429 Too Many Requests). Wait a few seconds and retry.';
    97→    case 500:
    98→    case 502:
    99→    case 503:
   100→    case 504:
   101→      return `OpenRouter upstream error (${status}). The model provider is having issues — retry in a moment. ${upstreamMessage}`.trim();
   102→    default:
   103→      return `OpenRouter API error (${status}): ${upstreamMessage || 'Unknown error'}`;
   104→  }
   105→}

   107→/**
   108→ * Call GLM via OpenRouter API — NO TOKEN LIMITS by default
   109→ * The model decides how many tokens it needs for full reasoning
   110→ *
   111→ * Error handling: surfaces a CLEAR, actionable message for the common
   112→ * failure modes (401 invalid key, 402 quota, 403 forbidden, 408/504 timeout,
   113→ * 429 rate-limit). The previous implementation threw a generic
   114→ * "OpenRouter API error (401): {\"error\":...}" string which was almost
   115→ * impossible for end users to act on.
   116→ */
   117→export async function callGLM(
   118→  messages: GLMMessage[],
   119→  config: GLMConfig
   120→): Promise<GLMResponse> {
   121→  const {
   122→    apiKey,
   123→    model,
   124→    temperature = 0.1,
   125→    // NO default maxTokens — let the model reason as long as it needs
   126→  } = config;
   127→
   128→  if (!apiKey) {
   129→    throw new Error('API key is required for GLM analysis');
   130→  }
   131→
   132→  // VPS KVM 2 (Hostinger, 8GB RAM) — no serverless limits. Allow GLM 5.2
   133→  // up to 5 minutes for deep analysis. User feedback:
   134→  //   "почём ты думаешь для ии достаточно 2 минут чтобы обойти полностью сайт
   135→  //    и протестировать все его точки на уязвимости"
   136→  // 2 minutes was cutting off deep multi-pass analysis. With 5 minutes,
   137→  // the model can:
   138→  //   - Parse large codebases (50K+ chars)
   139→  //   - Build complete source→dataflow→sink chains
   140→  //   - Construct concrete exploit scenarios
   141→  //   - Cross-reference with known exploits (DAO, bZx, etc.)
   142→  // If still running at 5 min, model is stuck — abort and use partial.
   143→  const callTimeout = config.timeoutMs || 60_000; // 60s — if GLM hasn't responded, abort and use static results
   144→  const controller = new AbortController();
   145→  const timeout = setTimeout(() => controller.abort(), callTimeout);
   146→  let response: Response;
   147→  try {
   148→    // Build request body
   149→    // GLM 5.2 reasoning mode: without max_tokens, the model generates
   150→    // infinite reasoning tokens and never stops. Set a generous default.
   151→    const requestBody: Record<string, unknown> = {
   152→      model: model || DEFAULT_MODEL,
   153→      messages,
   154→      temperature,
   155→    };
   156→    // max_tokens: 32768 for GLM models — MAXIMUM reasoning depth.
   157→    // Render has 15-minute timeout (vs Vercel Hobby 60s), so we can afford
   158→    // the full reasoning chain. This gives ~24,000 words of output — enough
   159→    // for thorough multi-finding analysis with full PoC outlines.
   160→    // No max_tokens limit — model decides when analysis is complete
   161→    // DeepSeek and other models keep their explicit limits from config
   162→
   163→    response = await fetch(OPENROUTER_API_URL, {
   164→      method: 'POST',
   165→      headers: {
   166→        'Content-Type': 'application/json',
   167→        'Authorization': `Bearer ${apiKey}`,
   168→        'HTTP-Referer': 'https://cryptosentinel.app',
   169→        'X-Title': 'CryptoSentinel',
   170→      },
   171→      body: JSON.stringify(requestBody),
   172→      signal: controller.signal,
   173→    });
   174→  } catch (fetchErr) {
   175→    clearTimeout(timeout);
   176→    if (fetchErr instanceof Error && fetchErr.name === 'AbortError') {
   177→      throw new Error(`OpenRouter API request timed out after ${callTimeout / 1000}s. The model may be overloaded — try again or switch to a faster model (e.g. GLM 4.7 Flash).`);
   178→    }
   179→    // Network-level failure (DNS, connection refused, etc.) — surface a
   180→    // distinct message so callers can distinguish from API errors.
   181→    throw new Error(`Network error reaching OpenRouter: ${fetchErr instanceof Error ? fetchErr.message : String(fetchErr)}`);
   182→  }
   183→  clearTimeout(timeout);
   184→
   185→  if (!response.ok) {
   186→    const errorBody = await response.text().catch(() => '');
   187→    throw new Error(formatOpenRouterError(response.status, errorBody, model || DEFAULT_MODEL));
   188→  }
   189→
   190→  const data = await response.json();
   191→
   192→  if (!data.choices || data.choices.length === 0) {
   193→    throw new Error('No response from model');
   194→  }
   195→
   196→  const message = data.choices[0].message;
   197→  // Some models (GLM 5.x with reasoning) return null content and put the
   198→  // answer in `reasoning`. Fall back to reasoning if content is empty.
   199→  let content = message.content;
   200→  if (!content && message.reasoning) {
   201→    // GLM 5.2 reasoning often wraps JSON in markdown fences — strip them
   202→    // before extracting the JSON array.
   203→    let reasoning = message.reasoning;
   204→    const mdMatch = reasoning.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
   205→    if (mdMatch) reasoning = mdMatch[1].trim();
   206→    // Try to extract JSON array from reasoning (some models put the answer there)
   207→    const jsonMatch = reasoning.match(/\[[\s\S]*\]/);
   208→    content = jsonMatch ? jsonMatch[0] : reasoning;
   209→  }
   210→  if (!content) {
   211→    // Model returned empty response
   212→    const finishReason = data.choices[0].finish_reason;
   213→    if (finishReason === 'length') {
   214→      throw new Error('Model ran out of tokens. This should not happen with unlimited tokens — the model may have hit a context window limit. Try with a shorter input.');
   215→    }
   216→    throw new Error('Model returned empty response');
   217→  }
   218→
   219→  return {
   220→    content,
   221→    model: data.model || model,
   222→    usage: data.usage,
   223→  };
   224→}

   441→ANALYSIS METHODOLOGY (Deep CodeQL-style reasoning chain):
   442→For EACH vulnerability, follow this FULL reasoning chain — do NOT skip steps:
   443→
   444→1. **SOURCE IDENTIFICATION**: Where does the attack surface originate? (msg.sender, function parameters, external calls, oracle data)
   445→2. **DATAFLOW TRACKING**: How does attacker-controlled data flow through the contract? Track EVERY variable assignment, function call, and state mutation in the path.
   446→3. **SINK REACHABILITY**: Does the tainted data reach a dangerous operation? (external call, state write, transfer, delegatecall)
   447→4. **SANITIZER CHECK**: Are there ANY guards? (require, onlyOwner, nonReentrant, bounds check, SafeMath). Check if they are sufficient or bypassable.
   448→5. **EXPLOIT CONSTRUCTION**: Construct a CONCRETE, REPRODUCIBLE attack scenario. Include specific function calls, parameter values, and state transitions.
   449→6. **IMPACT ASSESSMENT**: What is the MAXIMUM financial/systemic impact? Quantify in ETH/USD where possible. Use HackenProof severity tiers above.
   450→7. **CONFIDENCE SCORING**: Rate certainty on a 0.0-1.0 scale with detailed justification. If blockchain evidence confirms the vuln, confidence should be >= 0.90.
   451→8. **ON-CHAIN VERIFICATION**: If blockchain data is available, state whether the vulnerability is confirmed on-chain. Include specific transaction hashes or addresses if available.

   454→1. **Title**: Concise vulnerability name

   458→5. **Description**: DETAILED technical explanation following the FULL reasoning chain: source → dataflow → sink → sanitizer check → exploit → impact → on-chain evidence. Include HackenProof severity reasoning.
   459→6. **Validation Steps**: Step-by-step procedure to confirm the vulnerability (include on-chain verification steps when possible)
   460→7. **PoC Outline**: Detailed proof-of-concept attack contract with specific attack parameters
   461→8. **V1-V4 Scores**: Estimated scores for each validator (0.0-1.0) — higher when blockchain evidence confirms
   462→
   463→VALIDATOR SCORE CALIBRATION (V1-V4):
   464→Each validator measures a DIFFERENT line of evidence. Calibrate scores carefully:
   465→- **V1 (Symbolic Execution)**: Confidence that a symbolic-execution tool (e.g., Halmos, Manticore) would CONFIRM this vulnerability by reaching the vulnerable state through symbolic path exploration. 0.0 = no path exists; 1.0 = clearly reachable with concrete constraints.
   466→- **V2 (Fuzzing)**: Confidence that a fuzzer (e.g., Echidna, Medusa, Foundry invariant tests) would TRIGGER the bug within reasonable time. 0.0 = unreachable by random input; 1.0 = trivially triggered.
   467→- **V3 (Formal Verification)**: Confidence that a formal verifier (e.g., Certora, Scribble) would PROVE the violation against a suitable specification. 0.0 = no specification applies; 1.0 = violation directly contradicts a stated invariant.
   468→- **V4 (Economic)**: Confidence that the exploit is ECONOMICALLY viable — accounting for capital cost, MEV, slippage, gas, and competition. 0.0 = unprofitable or impractical; 1.0 = trivially profitable, no special conditions required.
   469→
   470→If you have NO basis to estimate a validator, set it to 0.5 (neutral). Do NOT set all four to the same value unless they truly coincide.
   471→9. **CWE**: The CWE or SWC ID for this vulnerability
   472→10. **BlockchainVerified**: true/false — whether on-chain evidence confirms this vulnerability
   473→11. **OnChainEvidence**: String describing any on-chain evidence (transaction hashes, addresses, patterns)
   474→
   475→CRITICAL EVIDENCE RULES (READ CAREFULLY — VIOLATING THESE INVALIDATES THE REPORT):
   476→
   477→You MUST distinguish three strictly different evidence tiers. Mixing them is the #1 quality failure mode.
   478→
   479→**TIER 1 — Confirmed configuration weakness / code smell (OBSERVATION only):**
   480→  Something is technically missing or non-ideal in the code, observable via static review, but you have NOT constructed a concrete exploit.
   481→  Examples: missing zero-check on a parameter, use of tx.origin, use of block.timestamp for randomness, missing event emission on a state change, missing timelock on admin functions.
   482→  These are NOT vulnerabilities by themselves. They are weaknesses that COULD become exploitable under specific conditions you have not yet demonstrated.
   483→  → Severity: LOW at most, unless paired with a Tier 2 finding.
   484→  → Title format: "Missing zero-check on <param>" / "Use of tx.origin in <function>" — NOT "Use of tx.origin enables auth bypass".
   485→  → Description format: "Observation: line N of <file> uses tx.origin for authorization. No concrete auth-bypass chain demonstrated — this is a defense-in-depth weakness."
   486→  → Do NOT speculate about exploit chains in the description. Save that for the validationSteps field.
   487→
   488→**TIER 2 — Confirmed vulnerability (PROVEN exploit chain):**
   489→  You have demonstrated a complete chain: a SOURCE where attacker-controlled data enters (msg.sender, function param, external call, oracle data), a DATAFLOW that propagates it without sufficient sanitization, and a SINK where it causes harm (external call, state write, transfer). All three must be concrete, not hypothetical.
   490→  Examples: reentrancy with a proven external call before state update, access control bypass with a proven attacker address calling a privileged function, integer overflow with concrete parameter values that trigger it.
   491→  → Severity: matches the actual financial impact of the proven chain (MEDIUM/HIGH/CRITICAL per HackenProof tiers above).
   492→  → Title format: "Reentrancy in <function> via <external call>" / "Access control bypass on <function>".
   493→  → Description format: "SOURCE: <concrete input>. DATAFLOW: <trace through code, line by line>. SINK: <concrete dangerous operation>. SANITIZER: <none found, or name and why insufficient>. EXPLOIT: <concrete attack scenario with function calls, params, state transitions>. IMPACT: <quantified financial/systemic impact>."
   494→
   495→**TIER 3 — Confirmed high-impact chain (PROVEN vuln + amplifying factor):**
   496→  A Tier 2 vulnerability combined with a Tier 1 weakness or external condition that materially amplifies its impact.
   497→  Example: proven reentrancy (Tier 2) + missing nonReentrant modifier elsewhere in flow (Tier 1) + flash-loan available to amplify capital (external condition) → critical fund-drain chain.
   498→  → Severity: HIGH or CRITICAL — but ONLY when both halves are proven.
   499→  → You MUST cite the Tier 2 finding by title and the Tier 1 finding by title.
   500→
   501→**FORBIDDEN LOGICAL LEAPS (these will cause your output to be rejected):**
   502→
   503→1. You may NOT turn a Tier 1 observation into a Tier 2 vulnerability.
   504→   - BAD: "Use of tx.origin → auth bypass possible → fund theft" (conflates tiers).
   505→   - GOOD: "Use of tx.origin in transfer() (Tier 1, LOW)" as a separate finding. If you also have a proven auth-bypass chain, add a SEPARATE Tier 3 finding that cites both.
   506→
   507→2. You may NOT use the words "possible", "could", "may", "might", "if attacked", "potentially" inside a Tier 2 or Tier 3 description.
   508→   - If you find yourself writing "an attacker could call...", you have NOT proven the chain. Downgrade to Tier 1 or omit the finding entirely (per the OUT OF SCOPE rule below).
   509→
   510→3. You may NOT claim a dataflow exists just because a pattern is "common" or "typical" in DeFi contracts.
   511→   - BAD: "Flash loans commonly manipulate AMM price oracles → price manipulation possible"
   512→   - GOOD: "Line N of getReserves() reads spot price from Uniswap V2 pair. Line N+10 of borrow() uses this price without TWAP. Attacker can flash-borrow 90% of pool, manipulate spot price by 35%, borrow 42% over collateralization, repay with profit. Concrete exploit attached."
   513→
   514→4. You may NOT claim "ACTIVE VALIDATION PASSED" or "On-chain evidence CONFIRMS" unless blockchain data actually validates the SPECIFIC claim.
   515→   - "Contract deployed at 0x... CONFIRMED" does NOT confirm a reentrancy exists.
   516→   - Match the evidence to the claim precisely.
   517→
   518→5. Severity MUST match the evidence tier:
   519→   - Tier 1 alone → LOW
   520→   - Tier 2 proven but limited impact (e.g. griefing without profit) → MEDIUM
   521→   - Tier 2 proven with fund-theft impact → HIGH or CRITICAL
   522→   - Tier 3 (Tier 2 + amplifying Tier 1 + economic viability) → CRITICAL
   523→
   524→6. You may NOT make absolutist claims about impact amplification.
   525→   - FORBIDDEN: "any tx.origin use makes the contract critical" (false — depends on whether tx.origin is actually used for authorization on a privileged function).
   526→   - CORRECT phrasing: "use of tx.origin for authorization on a privileged function MAY allow auth bypass IF an attacker can trick a privileged user into calling the attacker's contract; the actual exploitability depends on whether such a path exists."
   527→   - When discussing a Tier 1 weakness that COULD amplify a Tier 2, always qualify with the specific conditions required.
   528→
   529→7. You may NOT use the word "CONFIRMED" unless the validation scope is 'target'.
   530→   - "LAB-VALIDATED" means the exploit chain works in a local Foundry EVM. This proves technical viability ONLY, not that the deployed contract is exploitable (bytecode may differ, admin controls may exist on-chain, state may differ).
   531→   - "TARGET-VALIDATED" means the exploit was verified against the actual deployed contract (e.g. via a real on-chain call against the production address).
   532→   - "THEORETICAL" means no runtime validation was performed — static analysis / AI reasoning only.
   533→   - Use the matching label exactly. Never write "[ACTIVE VALIDATION PASSED]" or "Exploit succeeded on local EVM" without the LAB-VALIDATED qualifier — those phrases imply target-level confirmation.
   534→
   535→8. You MUST prove each step of a smart contract exploit chain — DO NOT ASSUME:
   536→   A finding is only Tier 2 if ALL of these are proven:
   537→     a. SOURCE: identify the exact function + parameter that attacker controls
   538→     b. REACHABILITY: prove the function is callable (not behind onlyOwner, not internal)
   539→     c. STATE: prove the contract state required for the exploit exists (e.g., contract holds ETH for reentrancy drain)
   540→     d. EXECUTION: construct a CONCRETE exploit with specific parameter values, not hypothetical
   541→     e. IMPACT: quantify the financial impact (ETH amount, token amount)
   542→     f. ECONOMIC VIABILITY: for flash loan attacks, prove the attack is profitable after gas + MEV costs
   543→
   544→   If ANY step is unproven, DOWNGRADE or OMIT.
   545→
   546→9. For reentrancy findings, you MUST:
   547→   - Identify the EXACT external call that enables reentrancy (line number, function name)
   548→   - Show the state update that happens AFTER the external call (the window of vulnerability)
   549→   - Specify what the attacker's fallback function does during re-entry
   550→   - Calculate how much ETH/tokens can be drained per tx
   551→   - State whether nonReentrant modifier is present on OTHER functions that could be used for cross-function reentrancy
   552→
   553→10. For oracle/price manipulation findings, you MUST:
   554→    - Identify the exact oracle source (Chainlink, Uniswap TWAP, spot price)
   555→    - Calculate the capital required to manipulate the price (flash loan amount)
   556→    - Show the exact profit calculation: manipulated_price * borrowed_amount - flash_loan_fee - gas
   557→    - State whether TWAP is used (if yes, time-weighted manipulation cost is exponential)
   558→
   559→**STYLE:**
   560→- Use precise, technical language. State exactly what you observed and exactly what you did NOT observe.
   561→- Hedging is REQUIRED when the evidence is incomplete. "Likely", "probable", "appears to" are correct language when you have not proven the chain.
   562→- Do NOT use marketing words ("devastating", "severe" as adjectives — only use the severity enum value).
   563→- For each finding, include a "PROOF CHAIN" section listing each step (a-f above) with PROVEN or UNPROVEN status.
   564→
   565→IMPORTANT — REPORTING DISCIPLINE:
   566→- If you are NOT certain a vulnerability exists OR cannot construct a concrete exploit, DO NOT report it as Tier 2 or Tier 3.
   567→- Either downgrade it to Tier 1 (with LOW severity) or omit it entirely (per the OUT OF SCOPE rule above).
   568→- It is BETTER to report 2 confirmed Tier 2 findings than 10 speculative ones.
   569→- A small number of high-confidence findings is FAR more valuable than a large number of speculative ones.
   570→
   571→DEEP REASONING INSTRUCTION:
   572→- You have UNLIMITED reasoning capacity. Think as deeply and thoroughly as needed.
   573→- Do NOT simplify or abbreviate your analysis for any reason.
   574→- Consider ALL edge cases, ALL interaction paths, ALL state transitions.
   575→- For DeFi protocols, analyze economic attack vectors (flash loans, MEV, sandwich attacks) in full detail.
   576→- Cross-reference with known exploits (DAO hack, bZx flash loan, Poly dragon, Cream finance, etc.)
   577→- Use formal verification reasoning where applicable (invariants, preconditions, postconditions).
   578→
   579→Be thorough and precise. Focus on real, exploitable vulnerabilities. Do NOT report false positives. Provide detailed validation steps that another security researcher could reproduce.
   580→
   581→Respond in JSON format as an array of vulnerabilities:
   582→[
   583→  {
   584→    "title": "...",
   585→    "type": "...",
   586→    "severity": "...",
   587→    "location": "...",
   588→    "description": "...",
   589→    "validationSteps": "...",
   590→    "pocOutline": "...",
   591→    "v1Symbolic": 0.0-1.0,
   592→    "v2Fuzzing": 0.0-1.0,
   593→    "v3Formal": 0.0-1.0,
   594→    "v4Economic": 0.0-1.0,
   595→    "cwe": "...",
   596→    "blockchainVerified": true/false,
   597→    "onChainEvidence": "..."
   598→  }
   599→]`;

   601→/**
   602→ * Analyze contract source code using GLM 5.2 — NO TOKEN LIMITS
   603→ */
   604→export async function analyzeWithGLM(
   605→  sourceCode: string,
   606→  contractName: string,
   607→  config: GLMConfig,
   608→  blockchainContext?: string
   609→): Promise<Array<{
   610→  title: string;
   611→  type: string;
   612→  severity: string;
   613→  location: string;
   614→  description: string;
   615→  validationSteps: string;
   616→  pocOutline: string;
   617→  v1Symbolic: number;
   618→  v2Fuzzing: number;
   619→  v3Formal: number;
   620→  v4Economic: number;
   621→  blockchainVerified?: boolean;
   622→  onChainEvidence?: string;
   623→}>> {
   624→  let userContent = `Analyze the following smart contract for vulnerabilities:\n\nContract: ${contractName}\n\`\`\`solidity\n${sourceCode}\n\`\`\`\n`;
   625→
   626→  // Add blockchain verification context if available
   627→  if (blockchainContext) {
   628→    userContent += `\n[BLOCKCHAIN-VERIFY] On-chain data available for this contract:\n${blockchainContext}\nUse this data to confirm or deny vulnerabilities. Update blockchainVerified and onChainEvidence fields accordingly.\n`;
   629→  }
   630→
   631→  userContent += `\nIdentify all vulnerabilities. Think DEEPLY — you have unlimited reasoning capacity. Your FINAL output MUST be a valid JSON array — no markdown, no prose after the array. Output ONLY the JSON array:\n[{"title":"...","type":"...","severity":"...","location":"...","description":"...","validationSteps":"...","pocOutline":"...","v1Symbolic":0.0,"v2Fuzzing":0.0,"v3Formal":0.0,"v4Economic":0.0,"cwe":"...","blockchainVerified":false,"onChainEvidence":""}]`;
   632→
   633→  const messages: GLMMessage[] = [
   634→    { role: 'system', content: VULN_ANALYSIS_SYSTEM_PROMPT },
   635→    { role: 'user', content: userContent },
   636→  ];
   637→
   638→  // Note: callGLM internally caps GLM models at 16384 tokens (see callGLM impl).
   639→  // This is sufficient for thorough multi-finding analysis. Previously the
   640→  // comments here claimed "unlimited reasoning" which was misleading — the
   641→  // cap existed but was hidden at 8192 (now 16384).
   642→  const response = await callGLM(messages, {
   643→    ...config,
   644→    temperature: 0.05, // Very low temperature for precise analysis
   645→    // No explicit maxTokens — callGLM applies the GLM default (16384)
   646→  });
   647→
   648→  // Parse JSON from response
   649→  try {
   650→    let jsonStr = response.content.trim();
   651→
   652→    // Strip markdown code fences (```json ... ``` or ``` ... ```)
   653→    // GLM 5.2 often wraps JSON responses in markdown blocks despite
   654→    // instructions not to. Without stripping, the JSON.parse fails and
   655→    // all findings are silently dropped.
   656→    // Try multiple approaches to extract JSON from markdown:
   657→    // 1. Match ```json ... ``` or ``` ... ```
   658→    // 2. If that fails, just strip all backtick sequences
   659→    const mdMatch1 = jsonStr.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
   660→    if (mdMatch1) {
   661→      jsonStr = mdMatch1[1].trim();
   662→    } else if (jsonStr.includes('```')) {
   663→      // Fallback: remove all ```json and ``` markers
   664→      jsonStr = jsonStr.replace(/```(?:json)?/gi, '').trim();
   665→    }
   666→
   667→    // Find the JSON array — look for the outermost [...] matching
   668→    const jsonMatch = jsonStr.match(/\[[\s\S]*\]/);
   669→    if (jsonMatch) {
   670→      jsonStr = jsonMatch[0];
   671→    }
   672→
   673→    const parsed = JSON.parse(jsonStr);
   674→    if (Array.isArray(parsed)) {
   675→      return parsed.map(v => ({
   676→        title: v.title || 'Unknown Vulnerability',
   677→        type: v.type || 'unknown',
   678→        severity: v.severity || 'medium',
   679→        location: v.location || `${contractName}:L1`,
   680→        description: v.description || 'No description provided',
   681→        validationSteps: v.validationSteps || 'Validation pending.',
   682→        pocOutline: v.pocOutline || '',
   683→        v1Symbolic: typeof v.v1Symbolic === 'number' ? v.v1Symbolic : 0.5,
   684→        v2Fuzzing: typeof v.v2Fuzzing === 'number' ? v.v2Fuzzing : 0.5,
   685→        v3Formal: typeof v.v3Formal === 'number' ? v.v3Formal : 0.5,
   686→        v4Economic: typeof v.v4Economic === 'number' ? v.v4Economic : 0,
   687→        blockchainVerified: typeof v.blockchainVerified === 'boolean' ? v.blockchainVerified : false,
   688→        onChainEvidence: v.onChainEvidence || '',
   689→      }));
   690→    }
   691→    return [];
   692→  } catch {
   693→    console.error('Failed to parse GLM response as JSON, attempting text extraction...');
   694→    return extractVulnsFromText(response.content, contractName);
   695→  }
   696→}

   698→/**
   699→ * Extract vulnerabilities from free-text AI response when JSON parsing fails.
   700→ */
   701→function extractVulnsFromText(text: string, contractName: string): Array<{
   702→  title: string; type: string; severity: string; location: string;
   703→  description: string; validationSteps: string; pocOutline: string;
   704→  v1Symbolic: number; v2Fuzzing: number; v3Formal: number; v4Economic: number;
   705→  blockchainVerified?: boolean; onChainEvidence?: string;
   706→}> {
   707→  if (!text || text.length < 50) return [];
   708→
   709→  const results: Array<{
   710→    title: string; type: string; severity: string; location: string;
   711→    description: string; validationSteps: string; pocOutline: string;
   712→    v1Symbolic: number; v2Fuzzing: number; v3Formal: number; v4Economic: number;
   713→    blockchainVerified?: boolean; onChainEvidence?: string;
   714→  }> = [];
   715→
   716→  const severityPattern = /\b(critical|high|medium|low)\b/gi;
   717→  const typePattern = /\b(reentrancy|oracle_manipulation|access_control|integer_overflow|flash_loan|front_running|delegatecall|storage_collision|unchecked_call|denial_of_service|business_logic|governance_hijack|info_exposure)\b/gi;
   718→
   719→  const lines = text.split('\n');
   720→  let currentVuln: { title: string; lines: string[] } | null = null;
   721→
   722→  for (const line of lines) {
   723→    const headerMatch = line.match(/^\s*\d+\.\s*\**(.+?)\**\s*$/);
   724→    if (headerMatch && headerMatch[1].length > 5) {
   725→      if (currentVuln && currentVuln.lines.length > 0) {
   726→        const desc = currentVuln.lines.join('\n').trim();
   727→        const detectedSev = desc.match(severityPattern)?.[0]?.toLowerCase() || 'medium';
   728→        const detectedType = desc.match(typePattern)?.[0]?.toLowerCase() || 'unknown';
   729→        results.push({
   730→          title: currentVuln.title,
   731→          type: detectedType,
   732→          severity: ['critical', 'high', 'medium', 'low'].includes(detectedSev) ? detectedSev : 'medium',
   733→          location: `${contractName}:L1`,
   734→          description: desc.slice(0, 2000),
   735→          validationSteps: 'Extracted from AI reasoning text. Run structured analysis for detailed validation.',
   736→          pocOutline: '',
   737→          v1Symbolic: 0.6, v2Fuzzing: 0.5, v3Formal: 0.4, v4Economic: 0.2,
   738→          blockchainVerified: false,
   739→          onChainEvidence: '',
   740→        });
   741→      }
   742→      currentVuln = { title: headerMatch[1].trim(), lines: [] };
   743→    } else if (currentVuln) {
   744→      currentVuln.lines.push(line);
   745→    }
   746→  }
   747→
   748→  if (currentVuln && currentVuln.lines.length > 0) {
   749→    const desc = currentVuln.lines.join('\n').trim();
   750→    const detectedSev = desc.match(severityPattern)?.[0]?.toLowerCase() || 'medium';
   751→    const detectedType = desc.match(typePattern)?.[0]?.toLowerCase() || 'unknown';
   752→    results.push({
   753→      title: currentVuln.title,
   754→      type: detectedType,
   755→      severity: ['critical', 'high', 'medium', 'low'].includes(detectedSev) ? detectedSev : 'medium',
   756→      location: `${contractName}:L1`,
   757→      description: desc.slice(0, 2000),
   758→      validationSteps: 'Extracted from AI reasoning text. Run structured analysis for detailed validation.',
   759→      pocOutline: '',
   760→      v1Symbolic: 0.6, v2Fuzzing: 0.5, v3Formal: 0.4, v4Economic: 0.2,
   761→      blockchainVerified: false,
   762→      onChainEvidence: '',
   763→    });
   764→  }
   765→
   766→  if (results.length === 0) {
   767→    const mentions = text.match(typePattern);
   768→    if (mentions) {
   769→      const uniqueTypes = [...new Set(mentions.map(m => m.toLowerCase()))];
   770→      for (const type of uniqueTypes.slice(0, 5)) {
   771→        results.push({
   772→          title: `AI detected: ${type.replace(/_/g, ' ')}`,
   773→          type,
   774→          severity: type === 'reentrancy' ? 'critical' : 'medium',
   775→          location: `${contractName}:L1`,
   776→          description: `AI reasoning identified a potential ${type.replace(/_/g, ' ')} vulnerability. See reasoning text for details.`,
   777→          validationSteps: 'Extracted from AI reasoning. Run structured analysis for validation.',
   778→          pocOutline: '',
   779→          v1Symbolic: 0.5, v2Fuzzing: 0.4, v3Formal: 0.3, v4Economic: 0.1,
   780→          blockchainVerified: false,
   781→          onChainEvidence: '',
   782→        });
   783→      }
   784→    }
   785→  }
   786→
   787→  const noisePatterns = /\b(already detected|false positive|not (?:really|a )?vulnerability|not applicable|not exploitable|correct behavior|no risk|safe|secure|benign)\b/i;
   788→  const filtered = results.filter(r => !noisePatterns.test(r.title) && !noisePatterns.test(r.description));
   789→
   790→  console.log(`Extracted ${filtered.length} vulnerabilities from reasoning text (filtered from ${results.length})`);
   791→  return filtered.slice(0, 8);
   792→}

   820→
   821→CRITICAL EVIDENCE RULES:

   823→- If the vulnerability was already proven (concrete source → dataflow → sink → exploit), use definitive language: "IS", "HAS", "CAUSES", "LEADS TO", "ALLOWS".
   824→- If the vulnerability is an OBSERVATION only (a missing check, a code smell, a configuration weakness), use observational language: "is missing", "does not include", "uses X pattern" — and DO NOT speculate about exploit chains in the description.

   833→
   834→  // Use DeepSeek for enhancement — lighter, faster task
   835→  const response = await callDeepSeek(messages, config);
   836→
   837→  return response.content;
   838→}

  1064→CRITICAL EVIDENCE RULES (READ CAREFULLY — VIOLATING THESE INVALIDATES THE REPORT):
