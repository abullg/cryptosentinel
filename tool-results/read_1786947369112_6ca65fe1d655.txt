   445→function detectLanguage(code: string): 'solidity' | 'move' | 'rust' | 'cairo' | 'web' {
   446→  if (code.includes('module ') && code.includes('fun ') && (code.includes('move_to') || code.includes('borrow_global') || code.includes('public entry fun') || code.includes('transfer::'))) return 'move';
   447→  if (code.includes('fn ') && (code.includes('solana') || code.includes('anchor') || code.includes('#[program]'))) return 'rust';
   448→  if (code.includes('fn ') && code.includes('felt252')) return 'cairo';
   449→  if (code.includes('pragma solidity') || code.includes('contract ') || code.includes('function ') && code.includes('uint256')) return 'solidity';
   450→  if (code.includes('<html') || code.includes('<script') || code.includes('fetch(') || code.includes('axios')) return 'web';
   451→  return 'solidity'; // default
   452→}
   453→
   454→function getPatterns(lang: string): Pattern[] {
   455→  switch (lang) {
   456→    case 'move': return MOVE_PATTERNS;
   457→    case 'rust': return RUST_PATTERNS;
   458→    case 'cairo': return CAIRO_PATTERNS;
   459→    case 'web': return WEB_PATTERNS;
   460→    default: return SOLIDITY_PATTERNS;
   461→  }
   462→}

   464→/**
   465→ * Run static pattern analysis on source code.
   466→ * Returns findings with confidence scores.
   467→ *
   468→ * Bug fixes vs prior version:
   469→ * 1. Dedup key now includes line number (previously: type:title collapsed
   470→ *    3 reentrancy sites in the same contract into 1 finding). Now we keep
   471→ *    all distinct locations of the same pattern.
   472→ * 2. Findings are sorted by confidence DESCENDING BEFORE the 15-finding
   473→ *    cap is applied. Previously, low-confidence patterns that matched
   474→ *    early in the source would consume the entire cap, silently dropping
   475→ *    high-confidence findings later in the file.
   476→ */
   477→export function runStaticScan(sourceCode: string, fileName: string): StaticFinding[] {
   478→  const lang = detectLanguage(sourceCode);
   479→  const patterns = getPatterns(lang);
   480→  const findings: StaticFinding[] = [];
   481→  // Dedup key now includes line number so multiple sites of the same
   482→  // pattern are reported as separate findings (e.g. 3 reentrancy sites
   483→  // produce 3 findings, not 1).
   484→  const seen = new Set<string>();
   485→
   486→  // Split into lines for location tracking
   487→  const lines = sourceCode.split('\n');
   488→
   489→  for (const pattern of patterns) {
   490→    // Reset regex lastIndex
   491→    pattern.regex.lastIndex = 0;
   492→
   493→    const matches = sourceCode.matchAll(pattern.regex);
   494→    for (const match of matches) {
   495→      // Find line number FIRST so we can include it in dedup key
   496→      const matchStart = match.index ?? 0;
   497→      const lineNumber = sourceCode.slice(0, matchStart).split('\n').length;
   498→
   499→      // Dedup: same pattern at the same line is the same finding,
   500→      // but same pattern at a DIFFERENT line is a distinct finding.
   501→      const dedupKey = `${pattern.type}:${pattern.title}:L${lineNumber}`;
   502→      if (seen.has(dedupKey)) continue;
   503→      seen.add(dedupKey);
   504→
   505→      // Extract code snippet (3 lines around match)
   506→      const snippetStart = Math.max(0, lineNumber - 2);
   507→      const snippetEnd = Math.min(lines.length, lineNumber + 2);
   508→      const codeSnippet = lines.slice(snippetStart, snippetEnd).join('\n');
   509→
   510→      findings.push({
   511→        type: pattern.type,
   512→        title: pattern.title,
   513→        severity: pattern.severity,
   514→        location: `${fileName}:L${lineNumber}`,
   515→        description: pattern.description(match, fileName) +
   516→          `\n\nMatched pattern: \`${match[0].slice(0, 80)}\`` +
   517→          `\nCode context:\n\`\`\`\n${codeSnippet}\n\`\`\``,
   518→        confidence: pattern.confidence,
   519→        v1Symbolic: pattern.v1,
   520→        v2Fuzzing: pattern.v2,
   521→        v3Formal: pattern.v3,
   522→        v4Economic: pattern.v4,
   523→        patternTag: pattern.type,
   524→      });
   525→    }
   526→  }
   527→
   528→  // Sort by confidence DESCENDING first — so the cap keeps the most
   529→  // important findings. Previously the cap was applied during the loop,
   530→  // which silently dropped later high-confidence findings.
   531→  findings.sort((a, b) => b.confidence - a.confidence);
   532→
   533→  // Apply the per-scan cap AFTER sorting (high-confidence findings win)
   534→  return findings.slice(0, 30);
   535→}
