    45→function getRpcUrl(chain: string): string {
    46→  const alchemy = process.env.ALCHEMY_API_KEY;
    47→  const infura = process.env.INFURA_PROJECT_ID;
    48→  if (alchemy) {
    49→    const alchemyMap: Record<string, string> = {
    50→      ethereum: `https://eth-mainnet.g.alchemy.com/v2/${alchemy}`,
    51→      polygon: `https://polygon-mainnet.g.alchemy.com/v2/${alchemy}`,
    52→      arbitrum: `https://arb-mainnet.g.alchemy.com/v2/${alchemy}`,
    53→      optimism: `https://opt-mainnet.g.alchemy.com/v2/${alchemy}`,
    54→      base: `https://base-mainnet.g.alchemy.com/v2/${alchemy}`,
    55→    };
    56→    if (alchemyMap[chain]) return alchemyMap[chain];
    57→  }
    58→  if (infura) {
    59→    const infuraMap: Record<string, string> = {
    60→      ethereum: `https://mainnet.infura.io/v3/${infura}`,
    61→      polygon: `https://polygon-mainnet.infura.io/v3/${infura}`,
    62→      arbitrum: `https://arbitrum-mainnet.infura.io/v3/${infura}`,

    65→    };
    66→    if (infuraMap[chain]) return infuraMap[chain];
    67→  }
    68→  return PUBLIC_RPCS[chain] || PUBLIC_RPCS.ethereum;

    71→/** Detect chain by trying each public RPC in parallel. 3s timeout each. */
    72→async function detectChain(address: string): Promise<string | null> {
    73→  const chains = Object.keys(PUBLIC_RPCS);
    74→  const checks = await Promise.all(chains.map(async (chain) => {
    75→    try {
    76→      const rpc = getRpcUrl(chain);
    77→      const code = execSync(`cast code ${address} --rpc-url ${rpc}`, {
    78→        timeout: 3_000, encoding: 'utf-8', stdio: 'pipe',
    79→      }).trim();
    80→      if (code.length > 4) return chain;
    81→    } catch {}
    82→    return null;
    83→  }));
    84→  return checks.find(c => c) || null;
    85→}

    87→interface OnChainState {
    88→  owner: string | null;

    95→async function queryOnChainState(address: string, chain: string): Promise<OnChainState> {
    96→  const rpc = getRpcUrl(chain);
    97→  const state: OnChainState = {
    98→    owner: null, paused: null, totalSupply: null, balance: null, bytecodeSize: 0,
    99→  };
   100→  try {
   101→    const code = execSync(`cast code ${address} --rpc-url ${rpc}`, {
   102→      timeout: 3_000, encoding: 'utf-8', stdio: 'pipe',
   103→    }).trim();
   104→    state.bytecodeSize = Math.floor((code.length - 2) / 2);
   105→  } catch {}
   106→  const callView = (selector: string): string | null => {
   107→    try {
   108→      const out = execSync(`cast call ${address} ${selector} --rpc-url ${rpc}`, {
   109→        timeout: 3_000, encoding: 'utf-8', stdio: 'pipe',
   110→      }).trim();
   111→      return out && out.startsWith('0x') && out.length >= 66 ? out : null;
   112→    } catch { return null; }
   113→  };
   114→  const ownerRaw = callView('0x8da5cb5b');
   115→  if (ownerRaw) { state.owner = '0x' + ownerRaw.slice(-40).toLowerCase(); }
   116→  if (!state.owner) {
   117→    const adminRaw = callView('0xf851a440');
   118→    if (adminRaw) { state.owner = '0x' + adminRaw.slice(-40).toLowerCase(); }
   119→  }
   120→  const pausedRaw = callView('0x5c975abb');
   121→  if (pausedRaw) state.paused = pausedRaw.slice(-2) === '01';
   122→  const supplyRaw = callView('0x18160ddd');
   123→  if (supplyRaw) state.totalSupply = BigInt(supplyRaw).toString();
   124→  try {
   125→    const bal = execSync(`cast balance ${address} --rpc-url ${rpc}`, {
   126→      timeout: 3_000, encoding: 'utf-8', stdio: 'pipe',
   127→    }).trim();
   128→    if (bal && /^\d+$/.test(bal)) state.balance = bal;
   129→  } catch {}
   130→  return state;
   131→}

   133→async function validateWithCastOnChain(
   134→  vuln: { type: string; title: string; severity: string; description: string; location: string },
   135→  contractAddress: string,
   136→  chain?: string,
   137→): Promise<ValidationResult> {
   138→  let actualChain = chain || '';
   139→  if (!actualChain) {
   140→    actualChain = (await detectChain(contractAddress)) || '';
   141→    if (!actualChain) {
   142→      return { confirmed: false, validationScope: 'theoretical',
   143→        evidence: `[TARGET-VALIDATION SKIPPED] Could not detect chain for ${contractAddress}.` };
   144→    }
   145→  }
   146→  let state: OnChainState;
   147→  try { state = await queryOnChainState(contractAddress, actualChain); }
   148→  catch (e: any) { return { confirmed: false, validationScope: 'theoretical',
   149→    evidence: `[TARGET-VALIDATION ERROR] ${String(e.message || e).slice(0, 200)}` }; }
   150→  const ev: string[] = [`[TARGET-VALIDATED] Queried ${contractAddress} on ${actualChain}. Bytecode: ${state.bytecodeSize} bytes.`];
   151→  if (state.owner) ev.push(`owner(): ${state.owner}`);
   152→  if (state.paused !== null) ev.push(`paused(): ${state.paused}`);
   153→  if (state.totalSupply) ev.push(`totalSupply(): ${state.totalSupply}`);
   154→  if (state.balance) ev.push(`ETH balance: ${state.balance} wei`);
   155→  let confirmed = false;
   156→  const vt = vuln.type.toLowerCase();
   157→  if ((vt === 'access_control' || vt === 'unauthorized_mint' || vt === 'governance_hijack') && state.owner) {
   158→    if (/^0x0{40}$/i.test(state.owner)) { ev.push(`[TARGET-CONFIRMS] Owner is zero address.`); confirmed = true; }
   159→    else { try { const oc = execSync(`cast code ${state.owner} --rpc-url ${getRpcUrl(actualChain)}`, { timeout: 3_000, encoding: 'utf-8', stdio: 'pipe' }).trim();
   160→      if (oc.length <= 4) { ev.push(`[TARGET-CONFIRMS] Owner ${state.owner} is EOA.`); confirmed = true; }
   161→      else { ev.push(`[TARGET-PARTIAL] Owner is contract (timelock/multisig).`); } } catch {} }
   162→  }
   163→  if ((vt === 'reentrancy' || vt === 'flash_loan' || vt === 'delegatecall') && state.balance) {
   164→    if (BigInt(state.balance) > 0n) { ev.push(`[TARGET-CONFIRMS] Contract holds ${state.balance} wei.`); confirmed = true; }
   165→  }
   166→  if ((vt === 'denial_of_service' || vt === 'permanent_pause') && state.paused !== null) {
   167→    if (state.paused) { ev.push(`[TARGET-CONFIRMS] Contract is paused.`); confirmed = true; }
   168→    else { ev.push(`[TARGET-REFUTES] Contract is NOT paused.`); }
   169→  }
   170→  return { confirmed, validationScope: 'target', evidence: ev.join('\n') };
   171→}

   177→function generatePoCTest(sourceCode: string, contractName: string, vuln: any): string {
   178→  const vt = vuln.type.toLowerCase();
   179→  const pocMap: Record<string, string> = {
   180→    reentrancy: `contract ReentrancyAttack { IVulnerable victim; function attack() external payable { victim.deposit{value: msg.value}(); victim.withdraw(); } receive() external payable { if (address(victim).balance >= 1 ether) victim.withdraw(); } }`,
   181→    access_control: `contract AccessControlAttack { function testUnauthorized() public { vm.prank(attacker); target.setOwner(attacker); } }`,
   182→    integer_overflow: `contract IntegerOverflowTest { function testOverflow() public { target.deposit(type(uint256).max); } }`,
   183→  };
   184→  const poc = pocMap[vt] || `contract PoCTest { function testVuln() public { /* generic test */ } }`;
   185→  return `// SPDX-License-Identifier: MIT\npragma solidity ^0.8.20;\nimport "forge-std/Test.sol";\ninterface IVulnerable { function deposit() external payable; function withdraw() external; }\n${poc}`;
   186→}

   188→async function validateWithFoundry(sourceCode: string, contractName: string, vuln: any): Promise<ValidationResult> {
   189→  const nameMatch = sourceCode.match(/contract\s+(\w+)\s*[{(:]/);
   190→  const actualName = contractName || nameMatch?.[1] || 'Target';
   191→  const tmpDir = `/tmp/foundry-${Date.now()}`;
   192→  try {
   193→    mkdirSync(join(tmpDir, 'src'), { recursive: true });
   194→    mkdirSync(join(tmpDir, 'test'), { recursive: true });
   195→    writeFileSync(join(tmpDir, 'src', `${actualName}.sol`), sourceCode);
   196→    writeFileSync(join(tmpDir, 'test', 'PoC.t.sol'), generatePoCTest(sourceCode, actualName, vuln));
   197→    writeFileSync(join(tmpDir, 'foundry.toml'), `[profile.default]\nsrc="src"\nout="out"\nlibs=["lib"]\nsolc_version="0.8.20"\noptimizer=true\n`);
   198→    const FORGE_STD_CACHE = '/opt/forge-lib/forge-std';
   199→    const forgeStdLink = join(tmpDir, 'lib', 'forge-std');
   200→    try {
   201→      mkdirSync(join(tmpDir, 'lib'), { recursive: true });
   202→      if (existsSync(FORGE_STD_CACHE)) { try { symlinkSync(FORGE_STD_CACHE, forgeStdLink, 'dir'); } catch {} }
   203→      else { try { execSync('git init && git add -A && git commit -m init', { cwd: tmpDir, timeout: 5_000, stdio: 'pipe' }); } catch {}
   204→        try { execSync('forge install foundry-rs/forge-std --no-commit --no-git', { cwd: tmpDir, timeout: 30_000, stdio: 'pipe' });
   205→          try { mkdirSync('/opt/forge-lib', { recursive: true }); execSync(`cp -r ${forgeStdLink} ${FORGE_STD_CACHE}`, { timeout: 10_000, stdio: 'pipe' }); } catch {} } catch {} }
   206→    } catch {}
   207→    const result = execSync('forge test -vvv 2>&1', { cwd: tmpDir, timeout: 30_000, encoding: 'utf-8', stdio: 'pipe' });
   208→    const passed = result.includes('[PASS]') || result.includes('SUCCESS');
   209→    const gasMatch = result.match(/(\d+)\s+gas/);
   210→    return passed ? { confirmed: true, validationScope: 'lab',
   211→      evidence: `Foundry PoC PASSED. Gas: ${gasMatch?.[1] || 'n/a'}. This confirms the exploit works in a local EVM, NOT that the deployed contract is exploitable.`,
   212→      testOutput: result.slice(0, 2000), gasUsed: gasMatch ? parseInt(gasMatch[1]) : undefined }
   213→      : { confirmed: false, validationScope: 'lab',
   214→        evidence: `Foundry PoC FAILED — exploit did not succeed under lab conditions.`,
   215→        testOutput: result.slice(0, 2000) };
   216→  } catch (e: any) { return { confirmed: false, validationScope: 'theoretical',
   217→    evidence: `Foundry error: ${String(e.message || e).slice(0, 300)}` }; }
   218→  finally { try { rmSync(tmpDir, { recursive: true, force: true }); } catch {} }
   219→}

   326→async function sendTestRequest(url: string, method: string, payload: string, param: string,
   327→  options: { timeoutMs?: number; followRedirect?: boolean } = {}): Promise<{ status: number; body: string; headers: Record<string, string>; finalUrl: string }> {
   328→  const { timeoutMs = 5_000, followRedirect = true } = options;
   329→  try {
   330→    const params = new URLSearchParams();
   331→    params.set(param, payload);
   332→    const init: RequestInit = { method: method.toUpperCase(), headers: { 'Content-Type': 'application/x-www-form-urlencoded', ...BROWSER_HEADERS },
   333→      signal: AbortSignal.timeout(timeoutMs), redirect: followRedirect ? 'follow' : 'manual' };
   334→    let targetUrl = url;
   335→    if (method.toUpperCase() === 'GET') { const u = new URL(url); u.searchParams.set(param, payload); targetUrl = u.toString(); }
   336→    else { init.body = params.toString(); }
   337→    const resp = await fetch(targetUrl, init);
   338→    const body = await resp.text();
   339→    const headers: Record<string, string> = {};
   340→    resp.headers.forEach((v, k) => { headers[k] = v; });
   341→    return { status: resp.status, body, headers, finalUrl: resp.url || targetUrl };
   342→  } catch { return { status: 0, body: '', headers: {}, finalUrl: url }; }
   343→}

   345→async function discoverTargetParameters(targetUrl: string): Promise<string[]> {
   346→  const discovered = new Set<string>();
   347→  try { const u = new URL(targetUrl); u.searchParams.forEach((_, key) => discovered.add(key)); } catch {}
   348→  try {
   349→    const resp = await fetch(targetUrl, { headers: BROWSER_HEADERS, signal: AbortSignal.timeout(5_000) });
   350→    const html = await resp.text();
   351→    const inputMatches = html.matchAll(/<input[^>]+name=["']([^"']+)["']/gi);
   352→    for (const m of inputMatches) { const name = m[1].trim(); if (name && !name.startsWith('_')) discovered.add(name); }
   353→    const hrefMatches = html.matchAll(/<a[^>]+href=["']([^"']+)["']/gi);
   354→    for (const m of hrefMatches) { try { const url = m[1].startsWith('http') ? new URL(m[1]) : new URL(m[1], targetUrl);
   355→      url.searchParams.forEach((_, key) => discovered.add(key)); } catch {} }
   356→  } catch {}
   357→  const skip = /^(csrf|_token|__|authenticity_token|nonce|session|sid)$/i;
   358→  const filtered = [...discovered].filter(p => !skip.test(p));
   359→  if (filtered.length === 0) filtered.push('q', 'search', 'query', 'id', 'url', 'redirect', 'next', 'return');
   360→  return filtered.slice(0, 3);
   361→}

   363→function extractParamFromVuln(vuln: { location: string; description: string }): string | null {
   364→  const text = `${vuln.location || ''} ${vuln.description || ''}`;
   365→  const m1 = text.match(/param(?:eter)?[=:]\s*([A-Za-z_][A-Za-z0-9_]*)/i);
   366→  if (m1) return m1[1];
   367→  const m2 = text.match(/[?&]([A-Za-z_][A-Za-z0-9_]*)=/);
   368→  if (m2) return m2[1];
   369→  const m3 = text.match(/(?:via|in the)\s*\?([A-Za-z_][A-Za-z0-9_]*)/i);
   370→  if (m3) return m3[1];
   371→  return null;
   372→}

   374→async function validateWebVulnerability(targetUrl: string, vuln: any): Promise<ValidationResult> {
   375→  const vulnType = vuln.type.toLowerCase();
   376→  let testSuite: WebTestPayload | null = null;
   377→  if (vulnType === 'xss') testSuite = XSS_PAYLOADS;
   378→  else if (vulnType === 'sql_injection') testSuite = SQLI_PAYLOADS;
   379→  else if (vulnType === 'ssrf') testSuite = SSRF_PAYLOADS;
   380→  else if (vulnType === 'open_redirect') testSuite = REDIRECT_PAYLOADS;
   381→  else if (vulnType === 'command_injection') testSuite = CMDI_PAYLOADS;
   382→  else if (vulnType === 'path_traversal') testSuite = PATH_TRAVERSAL_PAYLOADS;
   383→  else if (vulnType === 'cors_misconfig') testSuite = CORS_TEST;
   384→
   385→  // CSP-missing: single HEAD request, instant
   386→  if (vulnType === 'csp_missing' || vulnType === 'csrf' || vulnType === 'auth_bypass') {
   387→    try {
   388→      const resp = await fetch(targetUrl, { headers: BROWSER_HEADERS, signal: AbortSignal.timeout(5_000) });
   389→      const headers: Record<string, string> = {};
   390→      resp.headers.forEach((v, k) => { headers[k] = v; });
   391→      if (vulnType === 'csp_missing') {
   392→        const csp = headers['content-security-policy'] || '';
   393→        if (!csp) return { confirmed: true, validationScope: 'target',
   394→          evidence: `[TARGET-VALIDATED] CSP-missing confirmed: HTTP ${resp.status} from ${targetUrl} has no Content-Security-Policy header.`,
   395→          requestUrl: targetUrl, responseStatus: resp.status };
   396→        return { confirmed: false, validationScope: 'target',
   397→          evidence: `[TARGET-VALIDATED] CSP-missing REFUTED: CSP is present: ${csp.slice(0, 200)}`,
   398→          requestUrl: targetUrl, responseStatus: resp.status };
   399→      }
   400→    } catch (e: any) { return { confirmed: false, validationScope: 'theoretical',
   401→      evidence: `[TARGET-VALIDATION ERROR] ${String(e.message || e).slice(0, 200)}` }; }
   402→  }
   403→
   404→  // CORS: send Origin headers
   405→  if (vulnType === 'cors_misconfig') {
   406→    for (const origin of CORS_TEST.payloads) {
   407→      try {
   408→        const resp = await fetch(targetUrl, { headers: { ...BROWSER_HEADERS, 'Origin': origin }, signal: AbortSignal.timeout(5_000) });
   409→        const headers: Record<string, string> = {};
   410→        resp.headers.forEach((v, k) => { headers[k] = v; });
   411→        const body = await resp.text();
   412→        const fakeResp = { status: resp.status, body, headers };
   413→        if (CORS_TEST.check(fakeResp, origin)) return { confirmed: true, validationScope: 'target',
   414→          evidence: `[TARGET-VALIDATED] ${CORS_TEST.evidence(fakeResp, origin)}`, requestUrl: targetUrl, payload: origin };
   415→      } catch {}
   416→    }
   417→    return { confirmed: false, validationScope: 'target',
   418→      evidence: `[TARGET-VALIDATED] CORS test completed — ${CORS_TEST.payloads.length} origins tested. None triggered misconfiguration.`,
   419→      requestUrl: targetUrl };
   420→  }
   421→
   422→  // For payload-based tests (XSS, SQLi, etc): run ALL payloads in PARALLEL
   423→  const mentionedParam = extractParamFromVuln(vuln);
   424→  let candidateParams: string[] = [];
   425→  if (mentionedParam) candidateParams = [mentionedParam];
   426→  else candidateParams = await discoverTargetParameters(targetUrl);
   427→
   428→  if (!testSuite) {
   429→    // Unknown type — return lab scope (AI reasoning only)
   430→    return { confirmed: false, validationScope: 'lab',
   431→      evidence: `[LAB-VALIDATED] No active test suite for "${vulnType}". Finding based on AI reasoning only.` };
   432→  }
   433→
   434→  // Build ALL test combinations and run in PARALLEL
   435→  const needsNoRedirect = vulnType === 'open_redirect';
   436→  const allTests: Promise<{ confirmed: boolean; evidence: string; payload?: string; status?: number; body?: string }>[] = [];
   437→
   438→  for (const param of candidateParams) {
   439→    for (const payload of testSuite.payloads) {
   440→      for (const method of ['GET', 'POST']) {
   441→        allTests.push(
   442→          (async () => {
   443→            const resp = await sendTestRequest(targetUrl, method, payload, param, { followRedirect: !needsNoRedirect });
   444→            if (resp.status === 0) return { confirmed: false, evidence: '' };
   445→            if (testSuite!.check(resp, payload)) return {
   446→              confirmed: true,
   447→              evidence: `[TARGET-VALIDATED] ${testSuite!.evidence(resp, payload)} — ${method} ${targetUrl}?${param}=${payload} (HTTP ${resp.status})`,
   448→              payload, status: resp.status, body: resp.body.slice(0, 500),
   449→            };
   450→            // Open redirect: check Location header
   451→            if (needsNoRedirect && resp.status >= 300 && resp.status < 400) {
   452→              const location = resp.headers['location'] || '';
   453→              if (location.includes('evil.com')) return {
   454→                confirmed: true,
   455→                evidence: `[TARGET-VALIDATED] OPEN REDIRECT CONFIRMED: ${method} ${targetUrl}?${param}=${payload} → HTTP ${resp.status} Location: ${location}`,
   456→                payload, status: resp.status, body: resp.body.slice(0, 500),
   457→              };
   458→            }
   459→            return { confirmed: false, evidence: '' };
   460→          })()
   461→        );
   462→      }
   463→    }
   464→  }
   465→
   466→  // Wait for ALL tests in parallel
   467→  const results = await Promise.allSettled(allTests);
   468→  for (const r of results) {
   469→    if (r.status === 'fulfilled' && r.value.confirmed) {
   470→      return { confirmed: true, validationScope: 'target',
   471→        evidence: r.value.evidence, requestUrl: targetUrl, responseStatus: r.value.status, responseBody: r.value.body, payload: r.value.payload };
   472→    }
   473→  }
   474→
   475→  const totalTests = allTests.length;
   476→  return { confirmed: false, validationScope: 'target',
   477→    evidence: `[TARGET-VALIDATED] ${testSuite.name} test completed — ${totalTests} HTTP requests sent to ${targetUrl} across ${candidateParams.length} params (${candidateParams.join(', ')}). None confirmed.`,
   478→    requestUrl: targetUrl };
   479→}
   480→
   481→// ═══════════════════════════════════════════════════════════════════
   482→// MAIN ENTRY POINT
   483→// ═══════════════════════════════════════════════════════════════════
   484→
   485→export async function activelyValidate(
   486→  sourceCode: string,
   487→  contractName: string,
   488→  vuln: { type: string; title: string; severity: string; description: string; location: string },
   489→  _apiKey?: string,
   490→  _model?: string,
   491→): Promise<ValidationResult> {
   492→  const isSmartContract = sourceCode.includes('pragma solidity') ||
   493→    sourceCode.includes('contract ') ||
   494→    vuln.type === 'reentrancy' || vuln.type === 'access_control' ||
   495→    vuln.type === 'tx_origin' || vuln.type === 'integer_overflow' ||
   496→    vuln.type === 'unauthorized_mint' || vuln.type === 'flash_loan' ||
   497→    vuln.type === 'delegatecall' || vuln.type === 'governance_hijack';
   498→
   499→  const isWebVuln = vuln.type === 'xss' || vuln.type === 'sql_injection' ||
   500→    vuln.type === 'ssrf' || vuln.type === 'open_redirect' ||
   501→    vuln.type === 'command_injection' || vuln.type === 'path_traversal' ||
   502→    vuln.type === 'cors_misconfig' || vuln.type === 'business_logic' ||
   503→    vuln.type === 'csp_missing' || vuln.type === 'api_leak' ||
   504→    vuln.type === 'csrf' || vuln.type === 'auth_bypass';
   505→
   506→  if (isSmartContract) {
   507→    const addressMatch = sourceCode.match(/0x[0-9a-fA-F]{40}/) ||
   508→      vuln.location?.match(/0x[0-9a-fA-F]{40}/) ||
   509→      vuln.description?.match(/0x[0-9a-fA-F]{40}/);
   510→    const contractAddress = addressMatch?.[0];
   511→    if (contractAddress) {
   512→      try {
   513→        const targetResult = await validateWithCastOnChain(vuln, contractAddress);
   514→        if (targetResult.confirmed) return targetResult;
   515→        if (targetResult.evidence.includes('[TARGET-REFUTES]') || targetResult.evidence.includes('REFUTED')) return targetResult;
   516→      } catch {}
   517→    }
   518→    return validateWithFoundry(sourceCode, contractName, vuln);
   519→  }
   520→
   521→  if (isWebVuln) {
   522→    const urlMatch = vuln.location?.match(/https?:\/\/[^\s]+/) ||
   523→      vuln.description?.match(/https?:\/\/[^\s]+/) ||
   524→      sourceCode.match(/https?:\/\/[^\s]+/);
   525→    const targetUrl = urlMatch?.[0] || sourceCode;
   526→    if (!targetUrl.startsWith('http')) {
   527→      return { confirmed: false, validationScope: 'theoretical',
   528→        evidence: `Web vulnerability test skipped — no valid URL found. Finding remains at THEORETICAL validation level.` };
   529→    }
   530→    return validateWebVulnerability(targetUrl, vuln);
   531→  }
   532→
   533→  // Fallback: try both
   534→  if (sourceCode.includes('pragma solidity')) return validateWithFoundry(sourceCode, contractName, vuln);
   535→  const urlMatch = sourceCode.match(/https?:\/\/[^\s]+/);
   536→  if (urlMatch) return validateWebVulnerability(urlMatch[0], vuln);
   537→  return { confirmed: false, validationScope: 'theoretical',
   538→    evidence: `Could not determine test type for "${vuln.type}". Finding remains at THEORETICAL validation level.` };
   539→}
