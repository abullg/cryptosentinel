    59→// Persisted State Hook
    60→function usePersistedState<T>(key: string, defaultValue: T) {
    61→  const [state, setState] = useState<T>(() => {
    62→    if (typeof window !== 'undefined') {
    63→      try {
    64→        const saved = localStorage.getItem(key);
    65→        if (saved) return JSON.parse(saved);
    66→      } catch {}
    67→    }
    68→    return defaultValue;
    69→  });
    70→
    71→  // Debounced localStorage write — prevents UI jank from frequent writes during polling
    72→  const writeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    73→  useEffect(() => {
    74→    if (typeof window !== 'undefined') {
    75→      if (writeTimerRef.current) clearTimeout(writeTimerRef.current);
    76→      writeTimerRef.current = setTimeout(() => {
    77→        try { localStorage.setItem(key, JSON.stringify(state)); } catch {}
    78→      }, 500); // 500ms debounce
    79→    }
    80→    return () => { if (writeTimerRef.current) clearTimeout(writeTimerRef.current); };
    81→  }, [key, state]);
    82→
    83→  return [state, setState] as const;
    84→}

   568→  /** Fetch with timeout — prevents infinite loading if server hangs
   569→   *  Default 45s timeout for fetches — generous enough for slow networks.
   570→   *  Override for specific calls: 15s for 202 status, 10s for polls
   571→   *  IMPORTANT: Also respects the analysis AbortController signal */
   572→  const fetchWithTimeout = (url: string, options: RequestInit = {}, timeoutMs: number = 45_000): Promise<Response> => {
   573→    const controller = new AbortController();
   574→    const timer = setTimeout(() => controller.abort(), timeoutMs);
   575→    // Link to analysis abort: if analysis is aborted, this fetch aborts too.
   576→    // BUT only abort if the analysis signal is ACTUALLY aborted (not just
   577→    // because the ref changed). This prevents false-positive aborts when
   578→    // a previous analysis's ref is cleared during cleanup.
   579→    const analysisSignal = analysisAbortRef.current?.signal;
   580→    if (analysisSignal) {
   581→      if (analysisSignal.aborted) {
   582→        // Analysis already aborted — abort this fetch immediately
   583→        controller.abort();
   584→      } else {
   585→        // Only abort if the SAME signal fires (not a new one)
   586→        analysisSignal.addEventListener('abort', () => controller.abort(), { once: true });
   587→      }
   588→    }
   589→    return fetch(url, { ...options, signal: controller.signal }).finally(() => clearTimeout(timer));
   590→  };

   599→  /** UNIFIED SSE STREAMING ANALYSIS — THE DEFINITIVE FIX
   600→   *  Uses /api/analyze-stream for BOTH phases in ONE connection.
   601→   *  - Heartbeats every 5s keep the connection alive during long AI calls
   602→   *    during long GLM calls (which can take 60-180s).
   603→   *  - Static results appear immediately (phase1_complete event)
   604→   *  - AI results stream in as they're found (finding events)
   605→   *  - No gap between phases — same server instance
   606→   *  - Robust retry logic: 3 attempts with progressive backoff
   607→   *  - Sync /api/analyze-ai is now the SECONDARY fallback (it times out
   608→   *    at the server's request limit, so it only works for fast analyses). */
   609→  const runTwoPhaseAnalysis = async (projectId: string, code: string, contractNm: string, tUrl?: string, tType?: string, hpContext?: any) => {
   610→
   611→    // ═══════════════════════════════════════════════════════════════
   612→    // SSE STREAMING — PRIMARY path (most reliable for long AI calls)
   613→    // ═══════════════════════════════════════════════════════════════
   614→    // Serverless platforms cap HTTP requests at ~100s. The synchronous
   615→    // /api/analyze-ai endpoint can't fit a 30-180s GLM call inside that
   616→    // window. SSE streaming bypasses the limit because data flows
   617→    // continuously (heartbeats every 5s) — the proxy sees an active
   618→    // connection and doesn't kill it.
   619→    //
   620→    // ONE retry with 5s delay — this is enough for genuine network blips
   621→    // (TCP reset, brief packet loss) without forcing the user through
   622→    // 3 × 60s = 3+ minutes of "Connection failed" messages when the only
   623→    // issue was a slow AI call. With heartbeat timeout now at 180s, real
   624→    // timeouts will be rare.
   625→    const SSE_RETRIES = 1;
   626→    const SSE_RETRY_DELAYS = [5_000]; // 5s only
   627→
   628→    for (let attempt = 0; attempt < SSE_RETRIES; attempt++) {
   629→      throwIfAborted();
   630→      try {
   631→        addActivity('scan', attempt === 0 ? 'Starting deep analysis...' : `Retrying analysis (attempt ${attempt + 1}/${SSE_RETRIES})...`, 'running', 'Connecting', 5);
   632→
   633→        const result = await runSSEStream(projectId, code, contractNm, tUrl, tType, hpContext);
   634→        return result; // Success!
   635→
   636→      } catch (err: any) {
   637→        const errMsg = String(err);
   638→        if (errMsg.includes('AbortError') && analysisAbortRef.current?.signal.aborted) throw err;
   639→
   640→        const isNetworkError = errMsg.includes('Failed to fetch') || errMsg.includes('network error') || errMsg.includes('NetworkError') || errMsg.includes('fetch failed') || errMsg.includes('Load failed') || errMsg.includes('SSE heartbeat timeout');
   641→
   642→        if (isNetworkError && attempt < SSE_RETRIES - 1) {
   643→          addActivity('scan', `Connection failed, retrying in ${SSE_RETRY_DELAYS[attempt] / 1000}s...`, 'warning', 'Network blip', 10);
   644→          await new Promise(r => setTimeout(r, SSE_RETRY_DELAYS[attempt]));
   645→          continue;
   646→        }
   647→
   648→        // Non-network error or exhausted retries — try sync as last resort
   649→        if (!isNetworkError) throw err;
   650→      }
   651→    }
   652→
   653→    // ═══════════════════════════════════════════════════════════════
   654→    // FALLBACK: Synchronous approach (last resort — only works for
   655→    // fast analyses that complete within the server's request limit)
   656→    // ═══════════════════════════════════════════════════════════════
   657→    addActivity('scan', 'SSE streaming failed, trying sync analysis...', 'warning', 'Last resort', 15);
   658→    return await runSyncAnalysis(projectId, code, contractNm, tUrl, tType, hpContext);
   659→  };

   711→  /** SSE Stream handler — connects to /api/analyze-stream and processes events */
   712→  const runSSEStream = async (projectId: string, code: string, contractNm: string, tUrl?: string, tType?: string, hpContext?: any): Promise<{ contractId: string; auditId: string }> => {
   713→    return new Promise((resolve, reject) => {
   714→      let resolved = false;
   715→      let contractId = '';
   716→      let auditId = '';
   717→      let staticFindingsCount = 0;
   718→      let aiFindingsCount = 0;
   719→      let phase1Shown = false;
   720→
   721→      const doFetch = async () => {
   722→        try {
   723→          const res = await fetch('/api/analyze-stream', {
   724→            method: 'POST',
   725→            headers: { 'Content-Type': 'application/json' },
   726→            body: JSON.stringify({
   727→              projectId, sourceCode: code, contractName: contractNm,
   728→              targetUrl: tUrl, targetType: tType || targetType,
   729→              hackenproofContext: hpContext,
   730→            }),
   731→            signal: analysisAbortRef.current?.signal,
   732→          });
   733→
   734→          if (!res.ok) {
   735→            // If non-2xx, the SSE endpoint isn't available — fall back
   736→            const errText = await res.text().catch(() => '');
   737→            reject(new Error(`SSE endpoint: ${res.status} ${errText.slice(0, 100)}`));
   738→            return;
   739→          }
   740→
   741→          const reader = res.body?.getReader();
   742→          if (!reader) {
   743→            reject(new Error('No response body'));
   744→            return;
   745→          }
   746→
   747→          const decoder = new TextDecoder();
   748→          let buffer = '';
   749→          // Heartbeat timeout — server sends heartbeats every 5s. AI calls
   750→          // (GLM 5.2 deep analysis) can take up to 5 minutes on a real VPS
   751→          // (no serverless limits). During the AI call, Node.js event loop
   752→          // is busy with the OpenRouter HTTP request and heartbeats can be
   753→          // delayed. Set timeout to 10 min to accommodate:
   754→          //   - Normal AI call: 30-60s (heartbeats arrive, watchdog resets)
   755→          //   - Slow AI call: 60-180s (some heartbeats arrive, still resets)
   756→          //   - Deep reasoning on large codebase: 180-300s (still resets)
   757→          //   - Real dead connection: no heartbeats for 10 min → timeout
   758→          // Previous 180s was still too aggressive for deep reasoning on
   759→          // large smart contract audits with multi-pass AI analysis.
   760→          const HEARTBEAT_TIMEOUT_MS = 180_000; // 3 min — matches HARD_TIMEOUT
   761→          let lastEventTime = Date.now();
   762→
   763→          // Heartbeat watchdog — check every 30s.
   764→          const heartbeatWatchdog = setInterval(() => {
   765→            if (Date.now() - lastEventTime > HEARTBEAT_TIMEOUT_MS) {
   766→              clearInterval(heartbeatWatchdog);
   767→              if (!resolved) {
   768→                reader.cancel().catch(() => {});
   769→                reject(new Error('SSE heartbeat timeout — connection died'));
   770→              }
   771→            }
   772→          }, 15_000);
   773→
   774→          const processBuffer = () => {
   775→            // SSE format: "event: type\ndata: json\n\n"
   776→            const parts = buffer.split('\n\n');
   777→            buffer = parts.pop() || ''; // Keep incomplete chunk
   778→
   779→            for (const part of parts) {
   780→              if (!part.trim()) continue;
   781→              lastEventTime = Date.now();
   782→
   783→              let eventType = '';
   784→              let eventData = '';
   785→
   786→              for (const line of part.split('\n')) {
   787→                if (line.startsWith('event: ')) eventType = line.slice(7).trim();
   788→                else if (line.startsWith('data: ')) eventData = line.slice(6).trim();
   789→              }
   790→
   791→              if (!eventType || !eventData) continue;
   792→
   793→              try {
   794→                const data = JSON.parse(eventData);
   795→                handleSSEEvent(eventType, data);
   796→              } catch {
   797→                // Malformed data — ignore
   798→              }
   799→            }
   800→          };
   801→
   802→          const handleSSEEvent = (type: string, data: any) => {
   803→            switch (type) {
   804→              case 'heartbeat':
   805→                // Just keep connection alive — no action needed
   806→                break;
   807→
   808→              case 'progress':
   809→                if (data.step === 'connected') {
   810→                  // Stream established — server is alive and starting work.
   811→                  addActivity('scan', data.message || 'Stream connected', 'running', 'Connected', data.percent || 1);
   812→                } else if (data.step === 'static') {
   813→                  addActivity('scan', data.message || 'Running static analysis...', 'running', 'Phase 1', data.percent || 5);
   814→                } else if (data.step === 'ai_start' || data.step === 'ai_analysis') {
   815→                  if (!phase1Shown) phase1Shown = true;
   816→                  addActivity('scan', data.message || 'AI deep analysis...', 'running', 'Phase 2: AI', data.percent || 50);
   817→                } else if (data.step === 'blockchain' || data.step === 'websearch') {
   818→                  addActivity('scan', data.message, 'running', data.step, data.percent || 60);
   819→                } else if (data.step === 'enhancement' || data.step === 'onchain_verify') {
   820→                  addActivity('scan', data.message, 'running', data.step, data.percent || 90);
   821→                } else if (data.step === 'ai_done') {
   822→                  addActivity('scan', data.message, 'running', 'Processing', data.percent || 75);
   823→                }
   824→                break;
   825→
   826→              case 'phase1_complete': {
   827→                phase1Shown = true;
   828→                const findings = data.findings || [];
   829→                staticFindingsCount = findings.length;
   830→                contractId = data.contractId || '';
   831→                auditId = data.auditId || '';
   832→
   833→                // Show static findings immediately — only >=90% confidence
   834→                const highConfStaticFindings = onlyValidated(findings);
   835→                if (highConfStaticFindings.length > 0) {
   836→                  setVulns(prev => {
   837→                    const existingIds = new Set(prev.map(v => v.id));
   838→                    const newFindings = highConfStaticFindings.filter((f: any) => !existingIds.has(f.id));
   839→                    return [...newFindings, ...prev];
   840→                  });
   841→                }
   842→
   843→                if (data.needsAI) {
   844→                  addActivity('scan', `Static: ${findings.length} findings. Starting AI deep analysis...`, 'running', 'AI next', 40);
   845→                } else {
   846→                  addActivity('scan', `Static analysis: ${findings.length} findings (no AI key)`, 'success', 'Complete', 100);
   847→                  clearInterval(heartbeatWatchdog);
   848→                  resolved = true;
   849→                  resolve({ contractId, auditId });
   850→                }
   851→                break;
   852→              }
   853→
   854→              case 'finding': {
   855→                const vuln = data.vulnerability;
   856→                // Only add if BOTH conditions met:
   857→                //   1. confidence >= 90%
   858→                //   2. validationScope is 'target' or 'lab' (not theoretical/null)
   859→                // The server streams findings as they're found, but validation
   860→                // happens AFTER all findings are reported. So this check will
   861→                // usually FAIL here for the initial stream — the finding will
   862→                // be added later via the post-validation 'complete' event when
   863→                // its scope has been updated to 'target' or 'lab'.
   864→                if (vuln && (vuln.confidence || 0) >= 0.90) {
   865→                  aiFindingsCount++;
   866→                  setVulns(prev => {
   867→                    const existingIds = new Set(prev.map(v => v.id));
   868→                    if (existingIds.has(vuln.id)) return prev;
   869→                    return [vuln, ...prev];
   870→                  });
   871→                }
   872→                break;
   873→              }
   874→
   875→              case 'complete': {
   876→                const allFindings = data.findings || [];
   877→                // Merge any findings we haven't seen yet — only >=90% confidence
   878→                const highConfComplete = onlyValidated(allFindings);
   879→                if (highConfComplete.length > 0) {
   880→                  setVulns(prev => {
   881→                    const existingIds = new Set(prev.map(v => v.id));
   882→                    const newFindings = highConfComplete.filter((f: any) => !existingIds.has(f.id));
   883→                    if (newFindings.length === 0) return prev;
   884→                    return [...newFindings, ...prev];
   885→                  });
   886→                }
   887→                const total = highConfComplete.length;
   888→                const confirmed = highConfComplete.filter((f: any) => f.status === 'confirmed').length;
   889→                addActivity('scan', `Analysis complete: ${total} findings (${confirmed} confirmed)`, 'success', 'Full analysis done', 100);
   890→                clearInterval(heartbeatWatchdog);
   891→                resolved = true;
   892→                resolve({ contractId: contractId || data.contractId || '', auditId: auditId || data.auditId || '' });
   893→                break;
   894→              }
   895→
   896→              case 'error': {
   897→                const errorMsg = data.error || data.message || 'Unknown error';
   898→                addActivity('scan', `Analysis error: ${errorMsg.slice(0, 100)}`, 'warning', 'Static results available', 85);
   899→                // If we already have static results, resolve with what we have
   900→                if (staticFindingsCount > 0 || contractId) {
   901→                  clearInterval(heartbeatWatchdog);
   902→                  resolved = true;
   903→                  resolve({ contractId, auditId });
   904→                } else {
   905→                  clearInterval(heartbeatWatchdog);
   906→                  resolved = true;
   907→                  reject(new Error(errorMsg));
   908→                }
   909→                break;
   910→              }
   911→            }
   912→          };
   913→
   914→          // Read stream chunks
   915→          const readLoop = async () => {
   916→            try {
   917→              while (true) {
   918→                const { done, value } = await reader.read();
   919→                if (done) {
   920→                  // Stream ended — if not yet resolved, resolve with what we have
   921→                  clearInterval(heartbeatWatchdog);
   922→                  if (!resolved) {
   923→                    if (staticFindingsCount > 0 || contractId) {
   924→                      addActivity('scan', `Stream ended (${staticFindingsCount} static + ${aiFindingsCount} AI findings)`, 'success', 'Done', 100);
   925→                      resolved = true;
   926→                      resolve({ contractId, auditId });
   927→                    } else {
   928→                      resolved = true;
   929→                      reject(new Error('Stream ended with no results'));
   930→                    }
   931→                  }
   932→                  return;
   933→                }
   934→                buffer += decoder.decode(value, { stream: true });
   935→                processBuffer();

   937→            } catch (readErr: any) {
   938→              clearInterval(heartbeatWatchdog);
   939→              if (!resolved) {

   944→                  reject(new Error(`Stream read error: ${msg.slice(0, 80)}`));
   945→                }
   946→              }
   947→            }
   948→          };
   949→
   950→          readLoop();
   951→
   952→        } catch (fetchErr: any) {
   953→          if (!resolved) reject(fetchErr);
   954→        }
   955→      };
   956→
   957→      doFetch();
   958→    });
